## Goal

Ship the `/settings/about` page (version, git hash, runtime, licenses, manual update check, website link), the `electron-updater` integration (auto + manual checks, downloaded banner), and the user-facing telemetry switch (off by default; daily aggregate writes into `telemetry_local`).

## Architecture

- **About:** `src/components/settings/AboutTab.tsx` reads `__GIT_HASH__` (Vite-defined for renderer), `process.versions` (via IPC `app.runtimeInfo()`), `app.getVersion()`, and a build-time `build/licenses.json` (generated by `license-checker`). The page calls `ipc.update.checkManual()` and `shell.openExternal(url)` via existing IPC.
- **Auto-update:** `electron/update/updater.ts` wraps `electron-updater`'s `autoUpdater`. `initAutoUpdate()` (called only when `settings.update.autoCheck === true`) schedules a first check 60s after boot, then every 4h. Events bridge to the main window; the renderer shows a banner on `update-downloaded`.
- **Telemetry:** new `settings.telemetry.enabled` (default `false`). Phase 14 queue gets a new handler `kind='telemetry-aggregate'`; on enable, `scheduleDailyTelemetry()` enqueues the job at boot if today's row doesn't exist, else schedules for next 00:10 UTC. The handler aggregates yesterday's metrics from `ai_usage`, `clips`, `perf_samples` into `telemetry_local` (idempotent on `(day, metric)` unique index).

## Tech Stack

- `electron-updater` v6 (already in deps)
- `license-checker` (added in Plan 1)
- Vite `define` injection
- React 19 + i18n
- Phase 14 job queue (existing)

import { execSync } from 'node:child_process'

function gitHash(): string {
  if (process.env.NODE_ENV !== 'production') return 'dev'
  try {
    return execSync('git rev-parse --short HEAD').toString('utf8').trim()
  } catch {
    return 'dev'
  }
}

const HASH = JSON.stringify(gitHash())

export default defineConfig({
  main: {
    define: { __GIT_HASH__: HASH }
    // ... existing main config

export interface TelemetryAggregateInput {
  db: Database.Database
  day: string // 'YYYY-MM-DD' UTC
}

export async function handleTelemetryAggregate(input: TelemetryAggregateInput): Promise<void> {
  const { db, day } = input
  const dayStart = `${day}T00:00:00Z`
  const dayEnd = `${day}T23:59:59Z`

  const aiAgg = db
    .prepare(
      `SELECT
         COUNT(*) AS requests,
         COALESCE(SUM(prompt_tokens), 0) + COALESCE(SUM(completion_tokens), 0) AS total_tokens
       FROM ai_usage WHERE created_at >= ? AND created_at <= ?`
    )
    .get(dayStart, dayEnd) as { requests: number; total_tokens: number }

  const clipsCount = db
    .prepare(`SELECT COUNT(*) AS n FROM clips WHERE created_at >= ? AND created_at <= ?`)
    .get(dayStart, dayEnd) as { n: number } | undefined

  const perfCount = db
    .prepare(`SELECT COUNT(*) AS n FROM perf_samples WHERE ts >= ? AND ts <= ?`)
    .get(dayStart, dayEnd) as { n: number }

  const ups = db.prepare(
    `INSERT INTO telemetry_local (day, metric, value)
     VALUES (?, ?, ?)
     ON CONFLICT (day, metric) DO UPDATE SET value = excluded.value`
  )

  const tx = db.transaction(() => {
    ups.run(day, 'ai.requests', aiAgg.requests)
    ups.run(day, 'ai.tokens.total', aiAgg.total_tokens)
    ups.run(day, 'clips.created', clipsCount?.n ?? 0)
    ups.run(day, 'perf.samples', perfCount.n)
  })
  tx()
}

// electron/services/telemetry/scheduler.ts
import type { QueueStore } from '@/queue/store'

function utcDayOf(d: Date): string {
  return d.toISOString().slice(0, 10)
}

function nextRunAtMs(now: Date): number {
  const next = new Date(
    Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate(), 0, 10, 0, 0)
  )
  if (next.getTime() <= now.getTime()) next.setUTCDate(next.getUTCDate() + 1)
  return next.getTime() - now.getTime()
}

export function scheduleDailyTelemetry(deps: { store: QueueStore; now?: () => Date }): {
  stop: () => void
} {
  const now = deps.now ?? (() => new Date())
  let timer: NodeJS.Timeout | null = null

  function tick(): void {
    const d = now()
    const yesterday = new Date(d.getTime() - 86400_000)
    deps.store.enqueue('telemetry-aggregate', { day: utcDayOf(yesterday) })
    timer = setTimeout(tick, nextRunAtMs(now()))
  }

  // First run on enable: enqueue immediately for yesterday, then schedule.
  tick()
